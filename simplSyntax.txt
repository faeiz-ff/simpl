expressions
expression = <orTerm>

orTerm = <andTerm> ('|' <andTerm>)*
andTerm = equality ('&' <equality>)*
equality = <term> (('==' | '!=' | '>=' | '<=') <term>)*

term = <factor> (( '+' | '-' ) <factor>)*
factor = <primary> (( '*' | '/' ) <primary>)*

primary = <literalExpr> Literal
        | <groupingExpr> LPAREN could be id
        | <identifierExpr> ID
        | <functionCallExpr> ID LPAREN
        | <unaryExpr> BANG | PLUS | MINUS
        | <lambdaExpr> LPAREN ID 
        | nihil

identifierExpr = [name] ( '.' <identifierExpr>)?

unaryExpr = ( '!' | '+' | '-' ) <primary>

groupingExpr = '(' <expression> ')'

functionCallExpr = [callable] '(' (<expression> ( ',' <expression>)*)? ')'

lambdaExpr = '(' (<type> [param] ( ',' <type> [param] )* )? ')' <type> <blockStmt>
